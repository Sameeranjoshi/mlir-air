## Task 4: Implement `-air-to-csl` conversion pass

**Files:**
- Create: `mlir/include/air/Conversion/AIRToCSLPass.h`
- Create: `mlir/lib/Conversion/AIRToCSLPass.cpp`
- Modify: `mlir/include/air/Conversion/Passes.h`
- Modify: `mlir/lib/Conversion/Passes.cpp`
- Modify: `mlir/lib/Conversion/CMakeLists.txt`

**What this pass does:**
Reads `func.func @vecadd(%a, %b, %c)` containing `air.launch > air.segment > air.herd`. Produces a top-level `csl.wafer @vecadd` with three siblings: `csl.program @h` (herd body), `csl.layout @vecadd_layout` (1×1 placement), and `csl.host @vecadd` (memcpy + launch). Erases the original `func.func`.

**Kernel arg naming convention (V1):** Positional — `"arg0"`, `"arg1"`, ..., `"argN"`. The alias is the same as the var name.

**Direction classification:** Each herd kernel arg is classified as output if it is ever the `memref` of a `memref.store` inside the herd body; otherwise it is an input. Output args get `csl_host.memcpy_d2h` AFTER the launch; input args get `csl_host.memcpy_h2d` BEFORE the launch.

/// Return true if `arg` is used as the memref operand of a memref.store.
static bool isStoreTarget(BlockArgument arg) {
  for (Operation *user : arg.getUsers())
    if (auto storeOp = dyn_cast<memref::StoreOp>(user))
      if (storeOp.getMemref() == arg)
        return true;
  return false;
}

/// Lower a func.func with air.launch > air.herd to a top-level csl.wafer.
static LogicalResult lowerFuncToWafer(FuncOp func) {
  SmallVector<xilinx::air::LaunchOp> launches;
  func.walk([&](xilinx::air::LaunchOp op) { launches.push_back(op); });
  if (launches.empty())
    return success();

  MLIRContext *ctx = func.getContext();
  OpBuilder moduleBuilder(ctx);
  moduleBuilder.setInsertionPointAfter(func);

  for (auto launch : launches) {
    // Find the (single) herd under this launch.
    xilinx::air::HerdOp herd;
    launch.walk([&](xilinx::air::HerdOp h) { herd = h; });
    if (!herd)
      continue;

    Location loc = herd.getLoc();
    ArrayRef<BlockArgument> kArgs = herd.getKernelArguments();
    unsigned nArgs = kArgs.size();

    // Classify each kernel arg: output if it is a store target.
    SmallVector<bool, 4> isOut(nArgs, false);
    for (unsigned i = 0; i < nArgs; ++i)
      isOut[i] = isStoreTarget(kArgs[i]);

    // Naming: program = herd sym_name or "pe"; layout = "<func>_layout".
    std::string progName =
        herd.getSymName().value_or(StringRef("pe")).str();
    std::string layoutName = (func.getName() + "_layout").str();

    // ==================================================================
    // csl.wafer @<funcName> {arch = "wse3"}
    // ==================================================================
    auto archAttr = moduleBuilder.getDictionaryAttr(
        {moduleBuilder.getNamedAttr(
            "arch", moduleBuilder.getStringAttr("wse3"))});
    auto waferOp = moduleBuilder.create<xilinx::csl::WaferOp>(
        loc, func.getNameAttr(), archAttr);
    if (waferOp.getBody().empty())
      waferOp.getBody().emplaceBlock();
    Block *waferBlock = &waferOp.getBody().front();
    OpBuilder wb(ctx);
    wb.setInsertionPointToEnd(waferBlock);

    // ==================================================================
    // csl.program @<progName> { vars + func + exports }
    // ==================================================================
    auto progOp = wb.create<xilinx::csl::ProgramOp>(
        loc, wb.getStringAttr(progName));
    if (progOp.getBody().empty())
      progOp.getBody().emplaceBlock();
    Block *progBlock = &progOp.getBody().front();
    OpBuilder pb(ctx);
    pb.setInsertionPointToEnd(progBlock);

    // csl.var @argN : memref<...>
    SmallVector<Value, 4> varVals;
    for (unsigned i = 0; i < nArgs; ++i) {
      std::string vname = "arg" + std::to_string(i);
      auto varOp = pb.create<xilinx::csl::VarOp>(
          loc, kArgs[i].getType(), pb.getStringAttr(vname));
      varVals.push_back(varOp.getResult());
    }

    // csl.func @compute { <cloned herd body> }
    auto computeOp = pb.create<xilinx::csl::FuncOp>(
        loc, pb.getStringAttr("compute"));
    if (computeOp.getBody().empty())
      computeOp.getBody().emplaceBlock();
    Block *funcBlock = &computeOp.getBody().front();
    OpBuilder fb(ctx);
    fb.setInsertionPointToEnd(funcBlock);

    // Map herd kernel args → csl.var SSA values.
    IRMapping mapping;
    for (unsigned i = 0; i < nArgs; ++i)
      mapping.map(kArgs[i], varVals[i]);
    // Map tile-id and size block args to safe constants.
    Value zeroIdx = fb.create<arith::ConstantIndexOp>(loc, 0);
    Value oneIdx = fb.create<arith::ConstantIndexOp>(loc, 1);
    for (BlockArgument id : herd.getIds())
      mapping.map(id, zeroIdx);
    for (BlockArgument sz : herd.getSize())
      mapping.map(sz, oneIdx);
    // Clone all ops except herd_terminator.
    for (Operation &op : herd.getBody().front())
      if (!isa<xilinx::air::HerdTerminatorOp>(op))
        fb.clone(op, mapping);
    fb.create<xilinx::csl::ReturnOp>(loc);

    // csl.export @argN {alias = "argN"}
    for (unsigned i = 0; i < nArgs; ++i) {
      std::string vname = "arg" + std::to_string(i);
      auto expOp = pb.create<xilinx::csl::ExportOp>(
          loc, FlatSymbolRefAttr::get(ctx, vname));
      expOp->setAttr("alias", pb.getStringAttr(vname));
    }
    // csl.export @compute {kind = "func"}
    auto computeExp = pb.create<xilinx::csl::ExportOp>(
        loc, FlatSymbolRefAttr::get(ctx, "compute"));
    computeExp->setAttr("kind", pb.getStringAttr("func"));

    // ==================================================================
    // csl.layout {width=1, height=1} @<layoutName>
    // ==================================================================
    auto layoutAttrs = wb.getDictionaryAttr({
        wb.getNamedAttr("width",  wb.getI64IntegerAttr(1)),
        wb.getNamedAttr("height", wb.getI64IntegerAttr(1))
    });
    auto layoutOp = wb.create<xilinx::csl::LayoutOp>(
        loc, wb.getStringAttr(layoutName), layoutAttrs);
    if (layoutOp.getBody().empty())
      layoutOp.getBody().emplaceBlock();
    Block *layoutBlock = &layoutOp.getBody().front();
    OpBuilder lb(ctx);
    lb.setInsertionPointToEnd(layoutBlock);

    // csl_layout.place @<prog> at (0, 0)
    lb.create<xilinx::csl_layout::PlaceOp>(
        loc, FlatSymbolRefAttr::get(ctx, progName),
        lb.getI64IntegerAttr(0), lb.getI64IntegerAttr(0));

    // csl_layout.export "argN" from @<prog>::@argN
    for (unsigned i = 0; i < nArgs; ++i) {
      std::string vname = "arg" + std::to_string(i);
      SymbolRefAttr fromRef = SymbolRefAttr::get(
          ctx, progName, {FlatSymbolRefAttr::get(ctx, vname)});
      lb.create<xilinx::csl_layout::ExportOp>(
          loc, lb.getStringAttr(vname), fromRef, /*kind=*/StringAttr{});
    }
    // csl_layout.export "compute" from @<prog>::@compute {kind = "func"}
    SymbolRefAttr computeFrom = SymbolRefAttr::get(
        ctx, progName, {FlatSymbolRefAttr::get(ctx, "compute")});
    lb.create<xilinx::csl_layout::ExportOp>(
        loc, lb.getStringAttr("compute"), computeFrom,
        lb.getStringAttr("func"));

    // ==================================================================
    // csl.host @<funcName>(%args...) {layout = @<layout>}
    // ==================================================================
    auto hostAttrs = wb.getDictionaryAttr({
        wb.getNamedAttr("layout", FlatSymbolRefAttr::get(ctx, layoutName))
    });
    auto hostOp = wb.create<xilinx::csl::HostOp>(
        loc, wb.getStringAttr(func.getName()), hostAttrs);
    Block *hostBlock = &hostOp.getBody().emplaceBlock();
    for (unsigned i = 0; i < nArgs; ++i)
      hostBlock->addArgument(kArgs[i].getType(), loc);

    OpBuilder hb(ctx);
    hb.setInsertionPointToEnd(hostBlock);

    // memcpy_h2d for input args (before launch)
    for (unsigned i = 0; i < nArgs; ++i) {
      if (isOut[i]) continue;
      std::string vname = "arg" + std::to_string(i);
      SymbolRefAttr sym = SymbolRefAttr::get(
          ctx, layoutName, {FlatSymbolRefAttr::get(ctx, vname)});
      hb.create<xilinx::csl_host::MemcpyH2DOp>(
          loc, hostBlock->getArgument(i), sym,
          hb.getI64IntegerAttr(0), hb.getI64IntegerAttr(0),
          hb.getI64IntegerAttr(1), hb.getI64IntegerAttr(1));
    }
    // csl_host.launch @<layout>::@compute
    SymbolRefAttr launchSym = SymbolRefAttr::get(
        ctx, layoutName, {FlatSymbolRefAttr::get(ctx, "compute")});
    hb.create<xilinx::csl_host::LaunchOp>(loc, launchSym);
    // memcpy_d2h for output args (after launch)
    for (unsigned i = 0; i < nArgs; ++i) {
      if (!isOut[i]) continue;
      std::string vname = "arg" + std::to_string(i);
      SymbolRefAttr sym = SymbolRefAttr::get(
          ctx, layoutName, {FlatSymbolRefAttr::get(ctx, vname)});
      hb.create<xilinx::csl_host::MemcpyD2HOp>(
          loc, sym, hostBlock->getArgument(i),
          hb.getI64IntegerAttr(0), hb.getI64IntegerAttr(0),
          hb.getI64IntegerAttr(1), hb.getI64IntegerAttr(1));
    }
  }

  // Erase the original func.func (consumed; replaced by csl.wafer).
  func.erase();
  return success();
}

struct AIRToCSLPass
    : public PassWrapper<AIRToCSLPass, OperationPass<ModuleOp>> {

  MLIR_DEFINE_EXPLICIT_INTERNAL_INLINE_TYPE_ID(AIRToCSLPass)

  StringRef getArgument() const override { return "air-to-csl"; }
  StringRef getDescription() const override {
    return "Lower AIR dialect (1x1 herds) to CSL v2 wafer IR";
  }

  void getDependentDialects(DialectRegistry &registry) const override {
    registry.insert<xilinx::csl::CSLDialect,
                    xilinx::csl_layout::CSLLayoutDialect,
                    xilinx::csl_host::CSLHostDialect,
                    arith::ArithDialect,
                    memref::MemRefDialect,
                    scf::SCFDialect>();
  }

  void runOnOperation() override {
    ModuleOp module = getOperation();
    SmallVector<FuncOp> funcs;
    module.walk([&](FuncOp fn) { funcs.push_back(fn); });
    for (FuncOp fn : funcs) {
      if (failed(lowerFuncToWafer(fn))) {
        signalPassFailure();
        return;
      }
    }
  }
};

} // anonymous namespace

std::unique_ptr<mlir::OperationPass<mlir::ModuleOp>>
xilinx::air::createAIRToCSLPass() {
  return std::make_unique<AIRToCSLPass>();
}

module {

                    %c: memref<256xf32>) {

        : memref<256xf32>, memref<256xf32>, memref<256xf32> {

          }

        }

      }

  }
